"false" -}} {{- else -}} {{- $hideModel = site.Params.piReplay.hideModel | default false -}} {{- end -}} {{- $hideCwd := false -}} {{- with .Get "hideCwd" -}} {{- $hideCwd = ne (lower .) "false" -}} {{- else -}} {{- $hideCwd = site.Params.piReplay.hideCwd | default false -}} {{- end -}} {{- if not $data -}} {{- errorf "pi-replay: the `data` parameter is required, e.g. {{< pi-replay data=\"replay.json\" >}}" -}} {{- end -}} {{- $url := "" -}} {{- if or (strings.HasPrefix $data "http://") (strings.HasPrefix $data "https://") (strings.HasPrefix $data "/") -}} {{- $url = $data -}} {{- else -}} {{- with .Page.Resources.GetMatch $data -}} {{- $url = .RelPermalink -}} {{- else -}} {{- errorf "pi-replay: data file %q not found among the page resources of %s.
